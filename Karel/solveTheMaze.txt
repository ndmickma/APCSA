void solveTheMaze()
{
    // your code here
    turnRight();
    while(!onBeeper())
    {
        walk();
    }
}
void walk()
{
    if(leftIsClear())
    {
        turnLeft();
    }
    if(rightIsClear())
    {
        turnRight();
    }
    if(frontIsClear())
    {
        moveForward();
    }
    else
    {
        turnAround();
    }
}
