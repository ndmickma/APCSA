void saveTheFlower()
{
    // your code here
    moveForward();
    pickBeeper();
    repeat(4)
    {
        climbStair();
    }
    dropBeeper();
    moveForward();
    repeat(4)
    {
        goDownStair();
        
    }
}

void climbStair()
{
    turnLeft();
    repeat(2)
    {
        moveForward();
    }
    turnRight();
    moveForward();
}

void goDownStair()
{
    turnRight();
    repeat(2)
    {
        moveForward();
    }
    turnLeft();
    if(frontIsClear())
    {
        moveForward();
    }
}
