void walkTheLabyrinth()
{
    // your code here
    while(!onBeeper())
    {
        walk();
    }
    
}

void walk()
{
    if(frontIsClear())
    {
        moveForward();
    }
    if(rightIsClear())
    {
        turnRight();
        moveForward();
    }
    if(leftIsClear())
    {
        turnLeft();
        moveForward();
    }
}
