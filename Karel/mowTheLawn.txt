void mowTheLawn()
{
    // your code here
    repeat(3)
    {
        mowTwoRows();
    }
}

void mowTwoRows()
{
    while(beeperAhead())
    {
        moveForward();
        pickBeeper();
    }
    moveForward();
    turnLeft();
    moveForward();
    turnLeft();
    while(beeperAhead())
    {
        moveForward();
        pickBeeper();
    }
    moveForward();
    turnRight();
    moveForward();
    turnRight();
}
