void fetchTheStars()
{
    // your code here
    while(frontIsClear())
    {
        fetchStar();
    }
    
    
}
void fetchStar()
{
    turnLeft();
    while(frontIsClear())
    {
        moveForward();
    }
    pickBeeper();
    turnAround();
    
    comeDown();
    
}
void comeDown()
{
    while(frontIsClear())
    {
        moveForward();
    }
    dropBeeper();
}
